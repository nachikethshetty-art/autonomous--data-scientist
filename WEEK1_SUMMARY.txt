📈 WORKFLOW: UPLOAD → VALIDATE → DETECT PROBLEM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  User uploads CSV
         ↓
  Streamlit / FastAPI receives file
         ↓
  DataValidator processes
    • Parses CSV
    • Detects schema (auto-infer types)
    • Runs 5 quality checks
    • Stores in SQLite with unique job_id
         ↓
  Returns to user:
    ✅ Successful upload
    ✅ Schema (columns, types, nulls, cardinality)
    ✅ Validation results (5 checks passed/failed)
    ✅ Metadata (memory usage, duplicates, column counts)
         ↓
  AutoProblemDetector ready for next phase:
    • Heuristics: Identify target column candidates
    • LLM: Confirm with Gemini/Ollama
    • Determine: Classification or Regression
    • Confidence scoring + human review flag

💾 DATABASE SCHEMA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  File: data/jobs.db (SQLite)
  
  TABLE jobs
    ├─ job_id (TEXT, PRIMARY KEY)
    ├─ filename (TEXT)
    ├─ upload_timestamp (TIMESTAMP)
    ├─ file_path (TEXT)
    ├─ schema (TEXT - JSON)
    ├─ shape (TEXT - "100x5")
    ├─ rows (INTEGER)
    ├─ cols (INTEGER)
    ├─ status (TEXT - "pending|processing|complete")
    └─ metadata (TEXT - JSON)
    
  TABLE validations
    ├─ validation_id (INTEGER, PRIMARY KEY)
    ├─ job_id (TEXT, FOREIGN KEY)
    ├─ check_name (TEXT)
    ├─ passed (BOOLEAN)
    └─ message (TEXT)
